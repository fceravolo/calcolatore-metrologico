  <script>
    const plans = {
      BASE:       { price: 49, min: 1  },
      PRO:        { price: 69, min: 6  },
      ENTERPRISE: { price: 99, min: 16 }
    };

    let currentPlan = 'BASE';

    function selectPlan(el) {
      document.querySelectorAll('.plan-card').forEach(c => c.classList.remove('selected'));
      el.classList.add('selected');
      currentPlan = el.dataset.plan;
      const minQty = plans[currentPlan].min;
      const inp = document.getElementById('qty');
      inp.min = minQty;
      if ((parseInt(inp.value) || 0) < minQty) inp.value = minQty;
      document.getElementById('min-label').textContent = minQty;
      update();
    }

    function changeQty(delta) {
      const inp = document.getElementById('qty');
      const minQty = plans[currentPlan].min;
      inp.value = Math.max(minQty, (parseInt(inp.value) || minQty) + delta);
      update();
    }

    function clampAndUpdate() {
      const inp = document.getElementById('qty');
      const minQty = plans[currentPlan].min;
      if ((parseInt(inp.value) || 0) < minQty) inp.value = minQty;
      update();
    }

    function update() {
      const qty = parseInt(document.getElementById('qty').value) || plans[currentPlan].min;
      const planPrice = plans[currentPlan].price;
      const libAnnual = qty * 19;
      const totalMonth = planPrice + libAnnual / 12;
      const totalYear  = planPrice * 12 + libAnnual;

      document.getElementById('s-plan').textContent       = currentPlan;
      document.getElementById('s-plan-price').textContent = '€' + planPrice + ' / mese';
      document.getElementById('s-qty').textContent        = qty;
      document.getElementById('s-lib-price').textContent  = '€' + libAnnual + ' / anno';
      document.getElementById('s-total-month').textContent = '€' + totalMonth.toFixed(2);
      document.getElementById('s-total-year').textContent  = '€' + totalYear.toFixed(2);
    }

    update();
  </script>
